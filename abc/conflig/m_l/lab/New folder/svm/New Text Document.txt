# Load Iris dataset and filter for binary classification (setosa vs versicolor)
iris = load_iris()
X = iris.data
y = iris.target

# Filter for only two classes (0 and 1)
binary_mask = y < 2
X_binary = X[binary_mask]
y_binary = y[binary_mask]
feature_names = iris.feature_names
target_names = iris.target_names[:2]

print(f"Binary Dataset Shape: {X_binary.shape}")
print(f"Classes: {target_names[0]} (0) vs {target_names[1]} (1)")

# Split the data
X_train, X_test, y_train, y_test = train_test_split(
    X_binary, y_binary, test_size=0.3, random_state=42, stratify=y_binary
)

# Scale the features (important for SVM)
scaler = StandardScaler()
X_train_scaled = scaler.fit_transform(X_train)
X_test_scaled = scaler.transform(X_test)

# Create and train SVM with linear kernel
svm_linear = SVC(kernel='linear', random_state=42)
svm_linear.fit(X_train_scaled, y_train)

# Make predictions
y_pred = svm_linear.predict(X_test_scaled)

# Calculate accuracy
accuracy = accuracy_score(y_test, y_pred)
print(f"\nLinear SVM Accuracy: {accuracy:.4f}")

# Detailed classification report
print("\nClassification Report:")
print("="*50)
print(classification_report(y_test, y_pred, target_names=target_names))

# Confusion Matrix
cm = confusion_matrix(y_test, y_pred)
plt.figure(figsize=(6, 5))
sns.heatmap(cm, annot=True, fmt='d', cmap='Blues',
            xticklabels=target_names, yticklabels=target_names)
plt.title('Confusion Matrix - Linear SVM (Binary)')
plt.ylabel('Actual')
plt.xlabel('Predicted')
plt.show()

# Cross-validation scores
cv_scores = cross_val_score(svm_linear, X_train_scaled, y_train, cv=5)
print(f"\nCross-validation scores: {cv_scores}")
print(f"Average CV Score: {cv_scores.mean():.4f} (+/- {cv_scores.std()*2:.4f})")

# Get support vectors
print(f"\nNumber of Support Vectors: {len(svm_linear.support_)}")
print(f"Support Vector Indices: {svm_linear.support_}")

# Create synthetic 2D datasets for visualization
def plot_decision_boundary(X, y, model, title, ax, show_sv=True):
    """
    Plot decision boundary and support vectors for 2D data
    """
    # Create mesh
    h = 0.02
    x_min, x_max = X[:, 0].min() - 0.5, X[:, 0].max() + 0.5
    y_min, y_max = X[:, 1].min() - 0.5, X[:, 1].max() + 0.5
    xx, yy = np.meshgrid(np.arange(x_min, x_max, h),
                         np.arange(y_min, y_max, h))
    
    # Predict on mesh
    Z = model.predict(np.c_[xx.ravel(), yy.ravel()])
    Z = Z.reshape(xx.shape)
    
    # Plot decision boundary
    ax.contourf(xx, yy, Z, alpha=0.3, cmap='RdYlBu')
    ax.contour(xx, yy, Z, colors='black', linewidths=0.5, alpha=0.5)
    
    # Plot data points
    scatter = ax.scatter(X[:, 0], X[:, 1], c=y, cmap='RdYlBu', 
                         edgecolors='black', linewidth=1, s=50)
    
    # Plot support vectors
    if show_sv and hasattr(model, 'support_vectors_'):
        ax.scatter(model.support_vectors_[:, 0], model.support_vectors_[:, 1],
                  s=200, linewidth=1, facecolors='none', edgecolors='green',
                  label='Support Vectors')
    
    ax.set_xlabel('Feature 1')
    ax.set_ylabel('Feature 2')
    ax.set_title(title)
    ax.legend(loc='best')
    
    return scatter

# Practical prediction function with confidence
def svm_predict_with_confidence(features, model, scaler, threshold=0.5):
    """
    Make predictions with confidence based on distance from decision boundary
    
    Parameters:
    features: array-like, input features
    model: trained SVM model
    scaler: fitted StandardScaler
    threshold: confidence threshold
    """
    # Scale features
    features_scaled = scaler.transform([features])
    
    # Get prediction and decision function
    prediction = model.predict(features_scaled)[0]
    
    if hasattr(model, 'decision_function'):
        decision_value = model.decision_function(features_scaled)[0]
        
        # Convert decision value to confidence-like score (using sigmoid)
        confidence = 1 / (1 + np.exp(-np.abs(decision_value)))
        
        # Get distance from boundary
        distance = np.abs(decision_value)
    else:
        confidence = 0.5
        distance = 0
    
    result = {
        'prediction': prediction,
        'class_name': target_names_multi[prediction] if prediction < 3 else 'Unknown',
        'confidence': confidence,
        'distance_from_boundary': distance,
        'reliable': confidence >= threshold
    }
    
    return result
