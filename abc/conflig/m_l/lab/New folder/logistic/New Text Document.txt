1. Binary Classification using Logistic Regression (Titanic Dataset)
python
print("\n" + "="*70)
print("1. BINARY CLASSIFICATION - TITANIC SURVIVAL PREDICTION")
print("="*70)

# Load Titanic dataset
df = pd.read_csv("/content/Titanic-Dataset.csv")

print("Original Dataset Shape:", df.shape)
print("\nFirst few rows:")
print(df.head())

# Preprocess the data
df_binary = df.copy()

# Handle missing values
df_binary['Age'].fillna(df_binary['Age'].median(), inplace=True)
df_binary['Embarked'].fillna(df_binary['Embarked'].mode()[0], inplace=True)
df_binary['Fare'].fillna(df_binary['Fare'].median(), inplace=True)

# Drop unnecessary columns
df_binary.drop(['Cabin', 'Ticket', 'Name', 'PassengerId'], axis=1, inplace=True)

# Convert categorical variables
df_binary = pd.get_dummies(df_binary, columns=['Sex', 'Embarked'], drop_first=True)

print("\nProcessed Dataset Shape:", df_binary.shape)
print("Processed columns:", df_binary.columns.tolist())

# Prepare features and target
X_binary = df_binary.drop('Survived', axis=1)
y_binary = df_binary['Survived']

# Split the data
X_train, X_test, y_train, y_test = train_test_split(
    X_binary, y_binary, test_size=0.2, random_state=42, stratify=y_binary
)

# Scale the features
scaler = StandardScaler()
X_train_scaled = scaler.fit_transform(X_train)
X_test_scaled = scaler.transform(X_test)

# Create and train logistic regression model
log_reg_binary = LogisticRegression(random_state=42, max_iter=1000)
log_reg_binary.fit(X_train_scaled, y_train)

# Make predictions
y_pred_binary = log_reg_binary.predict(X_test_scaled)
y_pred_proba_binary = log_reg_binary.predict_proba(X_test_scaled)[:, 1]

# Calculate accuracy
accuracy = accuracy_score(y_test, y_pred_binary)
print(f"\nModel Accuracy: {accuracy:.4f}")

# Display feature importance (coefficients)
feature_importance = pd.DataFrame({
    'feature': X_binary.columns,
    'coefficient': log_reg_binary.coef_[0]
}).sort_values('coefficient', ascending=False)

print("\nFeature Coefficients (importance):")
print(feature_importance)

# Load Iris dataset
iris = load_iris()
X_iris = iris.data
y_iris = iris.target
iris_feature_names = iris.feature_names
iris_target_names = iris.target_names

print(f"Dataset Shape: {X_iris.shape}")
print(f"Number of classes: {len(np.unique(y_iris))}")
print(f"Class names: {iris_target_names}")
print(f"Feature names: {iris_feature_names}")

# Create DataFrame for better visualization
iris_df = pd.DataFrame(X_iris, columns=iris_feature_names)
iris_df['species'] = pd.Categorical.from_codes(y_iris, iris_target_names)
print("\nFirst few rows of Iris dataset:")
print(iris_df.head())

# Split the data
X_train_iris, X_test_iris, y_train_iris, y_test_iris = train_test_split(
    X_iris, y_iris, test_size=0.2, random_state=42, stratify=y_iris
)

# Scale the features
scaler_iris = StandardScaler()
X_train_iris_scaled = scaler_iris.fit_transform(X_train_iris)
X_test_iris_scaled = scaler_iris.transform(X_test_iris)

# Try different multiclass strategies
multiclass_models = {
    'One-vs-Rest (OvR)': LogisticRegression(multi_class='ovr', random_state=42, max_iter=1000),
    'Multinomial (Softmax)': LogisticRegression(multi_class='multinomial', random_state=42, max_iter=1000)
}

results_multiclass = {}

for model_name, model in multiclass_models.items():
    print(f"\n{'-'*50}")
    print(f"Model: {model_name}")
    print(f"{'-'*50}")
    
    # Train model
    model.fit(X_train_iris_scaled, y_train_iris)
    
    # Make predictions
    y_pred_iris = model.predict(X_test_iris_scaled)
    y_pred_proba_iris = model.predict_proba(X_test_iris_scaled)
    
    # Calculate accuracy
    accuracy = accuracy_score(y_test_iris, y_pred_iris)
    results_multiclass[model_name] = accuracy
    
    print(f"Accuracy: {accuracy:.4f}")
    
    # Confusion Matrix
    cm_iris = confusion_matrix(y_test_iris, y_pred_iris)
    print(f"\nConfusion Matrix:")
    print(cm_iris)
    
    # Classification Report
    print(f"\nClassification Report:")
    print(classification_report(y_test_iris, y_pred_iris, 
                              target_names=iris_target_names))

# Visualize comparison
plt.figure(figsize=(10, 5))

plt.subplot(1, 2, 1)
models = list(results_multiclass.keys())
accuracies = list(results_multiclass.values())
colors = ['skyblue', 'lightgreen']
bars = plt.bar(models, accuracies, color=colors, edgecolor='black')
plt.ylabel('Accuracy')
plt.title('Multiclass Classification - Model Comparison')
plt.ylim(0, 1)
for bar, acc in zip(bars, accuracies):
    plt.text(bar.get_x() + bar.get_width()/2, bar.get_height() + 0.01, 
             f'{acc:.4f}', ha='center', va='bottom')

# Visualize predictions distribution
plt.subplot(1, 2, 2)
# Use the multinomial model for demonstration
best_model = multiclass_models['Multinomial (Softmax)']
y_pred_proba_iris = best_model.predict_proba(X_test_iris_scaled)

for i, class_name in enumerate(iris_target_names):
    plt.hist(y_pred_proba_iris[:, i], alpha=0.5, label=class_name, bins=10)

plt.xlabel('Predicted Probability')
plt.ylabel('Frequency')
plt.title('Prediction Probability Distribution')
plt.legend()

plt.tight_layout()
plt.show()

def predict_titanic_passenger(pclass, age, sibsp, parch, fare, sex, embarked):
    """
    Predict survival for a Titanic passenger
    
    Parameters:
    pclass: 1, 2, or 3
    age: age in years
    sibsp: number of siblings/spouses aboard
    parch: number of parents/children aboard
    fare: ticket fare
    sex: 'male' or 'female'
    embarked: 'C', 'Q', or 'S'
    """
    
    # Create feature vector
    passenger = pd.DataFrame({
        'Pclass': [pclass],
        'Age': [age],
        'SibSp': [sibsp],
        'Parch': [parch],
        'Fare': [fare],
        'Sex_male': [1 if sex.lower() == 'male' else 0],
        'Embarked_Q': [1 if embarked.upper() == 'Q' else 0],
        'Embarked_S': [1 if embarked.upper() == 'S' else 0]
    })
    
    # Scale and predict
    passenger_scaled = scaler.transform(passenger)
    prediction = log_reg_binary.predict(passenger_scaled)[0]
    probability = log_reg_binary.predict_proba(passenger_scaled)[0]
    
    return prediction, probability

# Test the function with different scenarios
test_scenarios = [
    (1, 30, 0, 0, 100, 'female', 'S'),  # First class female
    (3, 25, 1, 0, 8.5, 'male', 'Q'),    # Third class male
    (2, 40, 0, 2, 25, 'female', 'C'),   # Second class female with children
]

print("\nTesting different passenger scenarios:")
for i, (pclass, age, sibsp, parch, fare, sex, embarked) in enumerate(test_scenarios):

    print(f"  Survival Probability: {prob[1]:.2%}")
