def plot_knn_decision_boundary(X, y, k_values, title_prefix, dataset_name):
    """
    Plot decision boundaries for different K values
    """
    n_rows = 2
    n_cols = 3
    fig, axes = plt.subplots(n_rows, n_cols, figsize=(18, 10))
    axes = axes.ravel()
    
    # Create mesh
    h = 0.02
    x_min, x_max = X[:, 0].min() - 0.5, X[:, 0].max() + 0.5
    y_min, y_max = X[:, 1].min() - 0.5, X[:, 1].max() + 0.5
    xx, yy = np.meshgrid(np.arange(x_min, x_max, h),
                         np.arange(y_min, y_max, h))
    
    # Create colormap
    cmap_light = ListedColormap(['#FFAAAA', '#AAFFAA', '#AAAAFF'])
    cmap_bold = ListedColormap(['#FF0000', '#00FF00', '#0000FF'])
    
    for idx, k in enumerate(k_values):
        if idx >= len(axes):
            break
            
        # Train KNN
        knn = KNeighborsClassifier(n_neighbors=k)
        knn.fit(X, y)
        
        # Predict on mesh
        Z = knn.predict(np.c_[xx.ravel(), yy.ravel()])
        Z = Z.reshape(xx.shape)
        
        # Plot
        axes[idx].contourf(xx, yy, Z, cmap=cmap_light, alpha=0.8)
        axes[idx].contour(xx, yy, Z, colors='black', linewidths=0.5, alpha=0.5)
        
        # Plot data points
        scatter = axes[idx].scatter(X[:, 0], X[:, 1], c=y, cmap=cmap_bold, 
                                   edgecolors='black', linewidth=1, s=50)
        
        axes[idx].set_xlabel('Feature 1')
        axes[idx].set_ylabel('Feature 2')
        axes[idx].set_title(f'{title_prefix} - K={k}')
        axes[idx].grid(True, alpha=0.3)
    
    # Hide empty subplots
    for idx in range(len(k_values), len(axes)):
        axes[idx].set_visible(False)
    
    plt.suptitle(f'KNN Decision Boundaries - {dataset_name}', fontsize=14, y=1.02)
    plt.tight_layout()
    plt.show()

Summary and Practical Applications
python
print("\n" + "="*70)
print("KNN - SUMMARY AND PRACTICAL APPLICATIONS")
print("="*70)

print("""
Key Characteristics of KNN:
--------------------------
1. Lazy Learning: No explicit training phase, stores all training data
2. Instance-based: Makes predictions based on similarity to training examples
3. Non-parametric: No assumptions about data distribution
4. Versatile: Can be used for both classification and regression

Effect of K Value:
----------------
• Small K (e.g., K=1): 
  - Low bias, high variance (overfitting)
  - Decision boundaries are complex and noisy
  
• Large K (e.g., K=15+):
  - High bias, low variance (underfitting)
  - Decision boundaries become smoother
  
• Optimal K:
  - Balances bias and variance
  - Typically found through cross-validation
  - Usually odd number to avoid ties

Distance Metrics Comparison:
--------------------------
• Euclidean (L2): Most common, works well for continuous features
• Manhattan (L1): More robust to outliers, good for high dimensions
• Chebyshev (L∞): Maximum coordinate difference
• Minkowski: Generalization of L1 and L2

Advantages of KNN:
----------------
✓ Simple and intuitive to understand
✓ No training phase (lazy learning)
✓ Naturally handles multi-class problems
✓ Can adapt to new data easily
✓ Works well with non-linear data

Limitations:
----------
✗ Computationally expensive for large datasets
✗ Sensitive to irrelevant features
✗ Requires feature scaling
✗ Memory intensive (stores all training data)
✗ Curse of dimensionality in high dimensions

Practical Guidelines:
-------------------
1. Always scale features before applying KNN
2. Use odd K values to avoid ties in classification
3. Choose K through cross-validation
4. Consider distance weighting for imbalanced data
5. For high-dimensional data, consider dimensionality reduction first

When to Use KNN:
--------------
✓ Small to medium-sized datasets
✓ When decision boundaries are expected to be irregular
✓ When interpretability is not the main concern
✓ For recommendation systems (similar users/items)
✓ When you need a quick, baseline model

# Practical prediction function
def knn_predict_with_info(features, model, scaler, k, X_train, y_train, feature_names=None):
    """
    Make predictions with additional information about nearest neighbors
    """
    # Scale features
    features_scaled = scaler.transform([features])
    
    # Get prediction and probabilities
    prediction = model.predict(features_scaled)[0]
    probabilities = model.predict_proba(features_scaled)[0]
    
    # Find nearest neighbors
    distances, indices = model.kneighbors(features_scaled)
    
    # Get neighbor information
    neighbors_info = []
    for i, (dist, idx) in enumerate(zip(distances[0], indices[0])):
        neighbors_info.append({
            'index': idx,
            'distance': dist,
            'true_label': y_train[idx],
            'class_name': target_names[y_train[idx]] if y_train[idx] < 3 else 'Unknown'
        })
    
    result = {
        'prediction': prediction,
        'class_name': target_names[prediction] if prediction < 3 else 'Unknown',
        'confidence': probabilities[prediction],
        'probabilities': dict(zip(target_names, probabilities)),
        'neighbors': neighbors_info,
        'avg_neighbor_distance': np.mean(distances[0])
    }
    
    return result
