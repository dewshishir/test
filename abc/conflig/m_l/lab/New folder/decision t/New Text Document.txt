def predict_iris(sepal_length, sepal_width, petal_length, petal_width):
    """
    Predict iris species based on measurements
    
    Parameters:
    sepal_length, sepal_width, petal_length, petal_width: float measurements in cm
    """
    # Create feature array
    features = np.array([[sepal_length, sepal_width, petal_length, petal_width]])
    
    # Make prediction
    prediction = pruned_tree.predict(features)[0]
    probabilities = pruned_tree.predict_proba(features)[0]
    
    # Get decision path
    leaf_id = pruned_tree.apply(features)[0]
    
    return {
        'species': target_names[prediction],
        'probabilities': dict(zip(target_names, probabilities)),
        'confidence': probabilities[prediction],
        'leaf_node': leaf_id
    }

print("""
Key Findings:
-------------
1. Decision Tree Structure:
   - Trees provide intuitive, interpretable decision rules
   - Can be visualized and understood by non-experts
   - Feature importance reveals which variables matter most

2. Performance Evaluation:
   - Training accuracy often high, but may indicate overfitting
   - Test accuracy provides realistic performance estimate
   - Cross-validation gives robust performance assessment

3. Splitting Criteria Comparison:
   - Gini and Entropy often produce similar results
   - Entropy may create slightly more balanced trees
   - Choice of criterion has minor impact on final performance

4. Pruning and Overfitting Control:
   - Unpruned trees tend to overfit training data
   - Parameters like max_depth, min_samples_split control complexity
   - Proper pruning improves generalization
   - Trade-off between bias and variance

5. Prediction Capabilities:
   - Trees can handle both binary and multiclass classification
   - Provide probability estimates for predictions
   - Decision path reveals reasoning behind each prediction
